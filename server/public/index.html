Integración en tu Pipeline

Ejemplo de GitHub Actions

jobs:
  deploy-storage:
    runs-on: ubuntu-latest
    steps:
      - name: Descargar binario
        run: |
          curl -L https://storage.example.com/latest/storage-linux-amd64 \
            -o storage-node
          chmod +x storage-node
      
      - name: Iniciar nodo
        run: |
          ./storage-node --port=${{ secrets.STORAGE_PORT }} \
                         --token=${{ secrets.STORAGE_TOKEN }}
